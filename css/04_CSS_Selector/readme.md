/* 
Types of selectors:
1. Element Selector
2. ID Selector
3. Class Selector
4. Grouping Selector
5. Universal Selector
*/

/* Element Selector */
p { color: blue; }

/* ID Selector */
#mainTitle { color: red; }

/* Class Selector */
.box { border: 2px solid black; }

/* Grouping Selector */
h1, h2, h3 { color: green; }

/* Universal Selector */
* { margin: 0; padding: 0; }
